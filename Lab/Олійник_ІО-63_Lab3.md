## 1. Мета лабораторної роботи

Засвоєння теоретичного матеріалу та набуття практичних навичок реалізації й застосування абстрактних структур даних **стек** (LIFO — «останнім прийшов, першим вийшов») та **черга** (FIFO — «першим прийшов, першим вийшов»), а також порівняння двох способів їхньої реалізації — на основі масиву фіксованого розміру та на основі однозв'язного списку.

## 2. Постановка задачі
1. Кожен варіант завдання визначає прикладну задачу (наприклад, перевірку збалансованості дужок, обчислення постфіксного виразу тощо) та структуру(и) даних, потрібну для її розв'язання (стек, чергу, або обидві).
2. Для розв'язання задачі написати **дві програми**, що реалізують ту саму структуру даних двома способами:
   1. на основі **одновимірного масиву фіксованого розміру** (з окремими змінними/індексами для вершини стека або голови й хвоста черги);
   2. на основі **однозв'язного списку** (динамічне виділення пам'яті: вузол зі значенням і вказівником на наступний вузол).

   Прикладна логіка (сама задача варіанта) в обох програмах має бути однаковою — відрізнятися має лише внутрішня реалізація стека/черги.
3. Обидві реалізації мають надавати єдиний узгоджений набір функцій, наприклад для стека: `push`, `pop`, `peek`, `is_empty`, `is_full`; для черги: `enqueue`, `dequeue`, `peek_front`, `is_empty`, `is_full`. Прикладна логіка (`main` і функції задачі) має викликати ці функції однаково незалежно від того, яка реалізація підключена — тобто дві програми відрізняються лише файлом/блоком, де реалізовані самі функції структури даних.
4. Розмір масиву для масивної реалізації взяти самостійно, достатнім для тестових прикладів (наприклад, 50–100 елементів), і оформити як іменовану константу.
5. **Масивна реалізація черги обов'язково має бути кільцевою (циклічною):** після заповнення останньої комірки масиву черга повинна продовжувати використовувати комірки, звільнені операціями `dequeue`, починаючи знову з початку масиву, а не вважати масив вичерпаним.
Так як мій номер у списку - 17, моє завдання має такий вигляд:

|№|завдання|
|---|---|
| 16 | Чергою змоделювати планування процесів методом кругового обслуговування (round robin) з фіксованим квантом часу: кожен процес має відому загальну тривалість виконання; на кожному кроці процес з початку черги виконується один квант, після чого (якщо не завершився) повертається в кінець черги. Вивести порядок завершення процесів. |
| 17 | Те саме, що варіант 16, але додатково визначити для кожного процесу його **час очікування** (сумарний час у черзі) та **час обороту** (від надходження до завершення). |

**1.(на основі одновимірного масиву фіксованого розміру)**
```
#%%
#include <stdio.h>
#define MAXSIZE 100

typedef struct {
    int id;
    int time;
    int remainingt;
    int arrivalt;
    int finisht;
    int waitt;
    int oborott;
} Process;

typedef struct {
    Process data[MAXSIZE];
    int front;
    int rear;
    int size;
} ArrayQueue;

void initQueue(ArrayQueue* q) {
    q->front = 0;
    q->rear = -1;
    q->size = 0;
}

int is_empty(ArrayQueue* q) {
    return q->size == 0;
}

int is_full(ArrayQueue* q) {
    return q->size == MAXSIZE;
}

void enqueue(ArrayQueue* q, Process p) {
    if (is_full(q)) {
        printf("CHERGA PEREPOVNENA\n"); 
        return;
    }
    q->rear = (q->rear + 1) % MAXSIZE;
    q->data[q->rear] = p;
    q->size++;
}

Process dequeue(ArrayQueue* q) {
    Process empty_p = {0, 0, 0, 0, 0, 0, 0};
    if (is_empty(q)) {
        printf("CHERGA POROJNYA\n"); 
        return empty_p;
    }
    Process p = q->data[q->front];
    q->front = (q->front + 1) % MAXSIZE;
    q->size--;
    return p;
}

int main() {
    ArrayQueue q;
    initQueue(&q);

    int n, quantum;
    printf("Kilzzevii masiv\n");
    printf("Kilkist procesiv: ");
    if (scanf("%d", &n) != 1 || n <= 0) return 1;

    printf("Kvant chasu: ");
    if (scanf("%d", &quantum) != 1 || quantum <= 0) return 1;

    Process processes[100];
    for (int i = 0; i < n; i++) {
        processes[i].id = i + 1;
        printf("chas vikonannya P%d: ", processes[i].id);
        scanf("%d", &processes[i].time);
        
        processes[i].remainingt = processes[i].time;
        processes[i].arrivalt = 0;
        processes[i].waitt = 0;
        processes[i].oborott = 0;
        
        enqueue(&q, processes[i]);
    }

    int current_time = 0;
    printf("Vikonannya\n");
    while (!is_empty(&q)) {
        Process p = dequeue(&q);

        if (p.remainingt > quantum) {
            current_time += quantum;
            p.remainingt -= quantum;
            printf("chas %3d: P%d vikonav %d kvant. Zalishilos: %d\n", current_time, p.id, quantum, p.remainingt);
            enqueue(&q, p);
        } else {
            current_time += p.remainingt;
            p.remainingt = 0;
            p.finisht = current_time;
            p.oborott = p.finisht - p.arrivalt;
            p.waitt = p.oborott - p.time;
            
            printf("chas %3d: P%d zaversheno. oborot: %d, ochikuvannya: %d\n", current_time, p.id, p.oborott, p.waitt);
            
            for (int i = 0; i < n; i++) {
                if (processes[i].id == p.id) processes[i] = p;
            }
        }
    }
    
    dequeue(&q); 

    
    printf("Resultat\n");
    double tw = 0, to = 0;
    for (int i = 0; i < n; i++) {
        tw += processes[i].waitt;
        to += processes[i].oborott;
        printf("proces P%d: chas vikonannya = %d, chas oborotu = %d, chas ochikuvannya = %d\n", 
               processes[i].id, processes[i].time, processes[i].oborott, processes[i].waitt);
    }
    return 0;
}
```

**2.(на основі однозв'язного списку)**

```
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int id;
    int time;
    int remainingt;
    int arrivalt;
    int finisht;
    int waitt;
    int oborott;
} Process;

typedef struct Node {
    Process data;
    struct Node* next;
} Node;

typedef struct {
    Node* front;
    Node* rear;
} ListQueue;

void initQueue(ListQueue* q) {
    q->front = NULL;
    q->rear = NULL;
}

int is_empty(ListQueue* q) {
    return q->front == NULL;
}


void enqueue(ListQueue* q, Process p) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    newNode->data = p;
    newNode->next = NULL;

    if (q->rear == NULL) {
        q->front = q->rear = newNode;
    } else {
        q->rear->next = newNode;
        q->rear = newNode;
    }
}

Process dequeue(ListQueue* q) {
    Process empty_p = {0, 0, 0, 0, 0, 0, 0};
    if (is_empty(q)) {
        printf("CHERGA POROJNYA\n");
        return empty_p;
    }

    Node* temp = q->front;
    Process p = temp->data;

    q->front = q->front->next;
    if (q->front == NULL) {
        q->rear = NULL;
    }

    free(temp);
    return p;
}

int main() {
    ListQueue q;
    initQueue(&q);

    int n, quantum;
    printf("Kilzzevii spisok\n");
    printf("Kilkist procesiv: ");
    if (scanf("%d", &n) != 1 || n <= 0) return 1;

            printf("chas %3d: P%d vikonav %d kvant. Zalishilos: %d\n", current_time, p.id, quantum, p.remainingt);
            enqueue(&q, p);
        } else {
            current_time += p.remainingt;
            p.remainingt = 0;
            p.finisht = current_time;

            p.waitt = p.oborott - p.time;

            printf("chas %3d: P%d zaversheno. oborot: %d, ochikuvannya: %d\n", current_time, p.id, p.oborott, p.waitt);

            for (int i = 0; i < n; i++) {
                if (processes[i].id == p.id) processes[i] = p;
            }
        }
```

## **Висновок:**
Метою лабораторної роботи було засвоєння теоретичного матеріалу та набуття практичних навичок реалізації й застосування абстрактних структур даних **стек** (LIFO — «останнім прийшов, першим вийшов») та **черга** (FIFO — «першим прийшов, першим вийшов»).  Порівнюючи їх, можна сказати,що однов`язний список, на відміну він масиву(який має чіткий ліміт (у мене це MAXSIZE у коді)), не має ліміту і обмежений тільки обсягом оперативної пам'яті. Обидві програми якщо черга порожня або ж переповнена відреагують і видадуть рядок згідно із ситуацією(або "черга переповнена", або "черга порожня") а не завершаться аварійно. 
Я думаю, що масив краще обирати тоді, коли вже знаєш мкасимальну к-сть елементів(через ліміт) з якими будеш працювати, бо він не буде навантажувати систему  частим виділенням пам'яті. А зв'язний список підійде тоді, коли кількість елементів невідома завчасно або може змінюватись(бо він більш гнучкий і немає жорстких обмежень у розмірі).
